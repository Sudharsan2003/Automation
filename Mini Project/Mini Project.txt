package proj;

import org.openqa.selenium.By;
import org.openqa.selenium.WebDriver;
import org.openqa.selenium.chrome.ChromeDriver;

public class PetStoreReptile {
	
	static WebDriver driver;
	static String url = "https://petstore.octoperf.com/actions/Catalog.action";
	static String reptile_xpath = "//div [@id = 'SidebarContent']/a[4]";
	static String product_xpath = "//tbody/tr[3]/td[1]/a[1]";
	static String addtocart_css = ".Button";
	static String Qty_txt_css = "input[name='EST-13']";
	static String update_xpath = "//input[contains(@value,'Cart')]";
	static String proceed_checkout_xpath = "//a[contains(text(),'Check')]";
	
	public static void verifyURL(String partofurl,String page) {
		String acturl = driver.getCurrentUrl();
		if(acturl.contains(partofurl)) {
			System.out.println(page + ": Pass");
		}else {
			System.out.println(page + ": Fail");
		}
	}
	
	public static void setUp() {
		driver = new ChromeDriver();
		driver.get(url);
	}

	public static void selectReptile() {
		driver.findElement(By.xpath(reptile_xpath)).click();
		verifyURL("categoryId=REPTILES","HomePage");
	}

	public static void selectProduct() {
		driver.findElement(By.xpath(product_xpath)).click();
		verifyURL("RP-LI-02","Product");
	}
	
	public static void addToCart() {
		driver.findElement(By.cssSelector(addtocart_css)).click();
		verifyURL("EST-13","Addtocart");
	}
	
	public static void updateCart() {
		driver.findElement(By.cssSelector(Qty_txt_css)).clear();
		driver.findElement(By.cssSelector(Qty_txt_css)).sendKeys("2");
		driver.findElement(By.xpath(update_xpath)).click();
		verifyURL("EST-13","Updatecart");
	}
	
	public static void checkout() {
		driver.findElement(By.xpath(proceed_checkout_xpath)).click();
		verifyURL("newOrderForm","Checkout");
	}
	
	public static void tearDown() {
		driver.quit();
	}

	public static void main(String[] args) {
		// TODO Auto-generated method stub
		setUp();
		selectReptile();
		selectProduct();
		addToCart();
		updateCart();
		checkout();
        tearDown();
	}

}
